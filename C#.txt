using System;
using System.Collections.Generic;

namespace KuantumAmbari
{
    // Özel exception
    public class KuantumCokusuException : Exception
    {
        public KuantumCokusuException(string id) 
            : base($"Kuantum çöküşü! Patlayan nesne ID: {id}") { }
    }

    // IKritik arayüzü
    public interface IKritik
    {
        void AcilDurumSogutmasi();
    }

    // Soyut ana sınıf
    public abstract class KuantumNesnesi
    {
        public string ID { get; set; }

        private double stabilite;
        public double Stabilite
        {
            get => stabilite;
            set
            {
                if (value > 100) stabilite = 100;
                else if (value < 0) stabilite = 0;
                else stabilite = value;

                if (stabilite <= 0)
                    throw new KuantumCokusuException(ID);
            }
        }

        private int tehlikeSeviyesi;
        public int TehlikeSeviyesi
        {
            get => tehlikeSeviyesi;
            set
            {
                if (value < 1) tehlikeSeviyesi = 1;
                else if (value > 10) tehlikeSeviyesi = 10;
                else tehlikeSeviyesi = value;
            }
        }

        public KuantumNesnesi(string id, double stabilite, int tehlike)
        {
            ID = id;
            Stabilite = stabilite;
            TehlikeSeviyesi = tehlike;
        }

        public abstract void AnalizEt();

        public string DurumBilgisi()
        {
            return $"ID: {ID} | Stabilite: {Stabilite:0.##} | Tehlike: {TehlikeSeviyesi}";
        }
    }

    // VeriPaketi
    public class VeriPaketi : KuantumNesnesi
    {
        public VeriPaketi(string id, double stabilite) : base(id, stabilite, 1) { }

        public override void AnalizEt()
        {
            Console.WriteLine("Veri içeriği okundu.");
            Stabilite -= 5;
        }
    }

    // KaranlikMadde
    public class KaranlikMadde : KuantumNesnesi, IKritik
    {
        public KaranlikMadde(string id, double stabilite, int tehlike) : base(id, stabilite, tehlike) { }

        public override void AnalizEt()
        {
            Console.WriteLine("Karanlık madde analiz ediliyor...");
            Stabilite -= 15;
        }

        public void AcilDurumSogutmasi()
        {
            Stabilite += 50;
            if (Stabilite > 100) Stabilite = 100;
            Console.WriteLine("Acil soğutma uygulandı.");
        }
    }

    // AntiMadde
    public class AntiMadde : KuantumNesnesi, IKritik
    {
        public AntiMadde(string id, double stabilite, int tehlike) : base(id, stabilite, tehlike) { }

        public override void AnalizEt()
        {
            Console.WriteLine("Evrenin dokusu titriyor...");
            Stabilite -= 25;
        }

        public void AcilDurumSogutmasi()
        {
            Stabilite += 50;
            if (Stabilite > 100) Stabilite = 100;
            Console.WriteLine("Acil soğutma uygulandı.");
        }
    }

    class Program
    {
        static Random rnd = new Random();
        static List<KuantumNesnesi> envanter = new List<KuantumNesnesi>();

        static void Main(string[] args)
        {
            try
            {
                while (true)
                {
                    Console.WriteLine("\nKUANTUM AMBARI KONTROL PANELİ");
                    Console.WriteLine("1. Yeni Nesne Ekle");
                    Console.WriteLine("2. Tüm Envanteri Listele");
                    Console.WriteLine("3. Nesneyi Analiz Et (ID ile)");
                    Console.WriteLine("4. Acil Durum Soğutması Yap (ID ile)");
                    Console.WriteLine("5. Çıkış");
                    Console.Write("Seçiminiz: ");
                    var sec = Console.ReadLine();

                    switch (sec)
                    {
                        case "1":
                            YeniNesneUret();
                            break;
                        case "2":
                            Listele();
                            break;
                        case "3":
                            AnalizIste();
                            break;
                        case "4":
                            SogutmaIste();
                            break;
                        case "5":
                            return;
                        default:
                            Console.WriteLine("Geçersiz seçim.");
                            break;
                    }
                }
            }
            catch (KuantumCokusuException ex)
            {
                Console.WriteLine("\n\nSİSTEM ÇÖKTÜ! TAHLİYE BAŞLATILIYOR...");
                Console.WriteLine(ex.Message);
            }
        }

        static void YeniNesneUret()
        {
            int tip = rnd.Next(1, 4); // 1-3
            string id = Guid.NewGuid().ToString().Substring(0, 8);
            double stabilite = rnd.Next(30, 101); // başlangıç stabilitesi
            switch (tip)
            {
                case 1:
                    envanter.Add(new VeriPaketi(id, stabilite));
                    Console.WriteLine($"VeriPaketi eklendi. ID: {id}");
                    break;
                case 2:
                    envanter.Add(new KaranlikMadde(id, stabilite, rnd.Next(3, 8)));
                    Console.WriteLine($"KaranlıkMadde eklendi. ID: {id}");
                    break;
                case 3:
                    envanter.Add(new AntiMadde(id, stabilite, rnd.Next(7, 10)));
                    Console.WriteLine($"AntiMadde eklendi. ID: {id}");
                    break;
            }
        }

        static void Listele()
        {
            if (envanter.Count == 0) { Console.WriteLine("Envanter boş."); return; }
            foreach (var n in envanter)
            {
                Console.WriteLine(n.DurumBilgisi());
            }
        }

        static KuantumNesnesi Bul(string id)
        {
            foreach (var n in envanter) if (n.ID == id) return n;
            return null;
        }

        static void AnalizIste()
        {
            Console.Write("Analiz edilecek nesnenin ID'si: ");
            var id = Console.ReadLine();
            var nesne = Bul(id);
            if (nesne == null) { Console.WriteLine("ID bulunamadı."); return; }
            nesne.AnalizEt();
            Console.WriteLine("Analiz sonrası: " + nesne.DurumBilgisi());
        }

        static void SogutmaIste()
        {
            Console.Write("Soğutulacak nesnenin ID'si: ");
            var id = Console.ReadLine();
            var nesne = Bul(id);
            if (nesne == null) { Console.WriteLine("ID bulunamadı."); return; }

            if (nesne is IKritik kritik)
            {
                kritik.AcilDurumSogutmasi();
                Console.WriteLine("Soğutma sonrası: " + nesne.DurumBilgisi());
            }
            else
            {
                Console.WriteLine("Bu nesne soğutulamaz!");
            }
        }
    }
}
